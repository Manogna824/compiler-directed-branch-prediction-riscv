module instr_mem (
    input  wire [31:0] addr,
    output wire [31:0] instr
);
    reg [31:0] mem [0:255];
    integer k;
    initial begin
        for (k = 0; k < 256; k = k + 1)
            mem[k] = 32'h00000013; // NOP
 
        mem[ 0] = 32'h10000293; // addi x5,x0,256   base address of string
        mem[ 1] = 32'h00000493; // addi x9,x0,0     length counter = 0
        // LOOP at PC=8 (idx 2)
        mem[ 2] = 32'h0002A183; // lw x3,0(x5)      load current char
        mem[ 3] = 32'h00018863; // beq x3,x0,+16    if null → jump to DONE (hint=0)
        mem[ 4] = 32'h00148493; // addi x9,x9,1     length++
        mem[ 5] = 32'h00428293; // addi x5,x5,4     advance pointer
        mem[ 6] = 32'hFE0198E3; // bne x3,x0,-16    loop back  (hint=0, funct3=1)
        // DONE at PC=28 (idx 7)
        mem[ 7] = 32'h00048513; // addi x10,x9,0    x10 = result
        mem[ 8] = 32'h00000063; // beq x0,x0,0      HALT
    end
    assign instr = mem[addr[31:2]];
endmodule
